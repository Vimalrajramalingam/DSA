class Solution {
    int count=0;
    public int findTargetSumWays(int[] nums, int target) {
        back(nums,0,0,target);
        return count;        
    }
    public void back(int nums[],int index,int sum,int target){
        if(index==nums.length){
            if(sum==target){
                count++;
            }
            return;
        }
        back(nums,index+1,sum+nums[index],target);
        back(nums,index+1,sum-nums[index],target);

    }
}
