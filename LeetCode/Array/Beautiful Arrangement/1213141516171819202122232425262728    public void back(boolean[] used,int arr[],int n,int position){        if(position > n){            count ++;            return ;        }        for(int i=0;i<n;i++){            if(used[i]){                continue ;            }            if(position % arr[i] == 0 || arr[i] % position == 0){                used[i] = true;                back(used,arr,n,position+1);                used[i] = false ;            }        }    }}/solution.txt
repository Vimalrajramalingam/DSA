    public void back(boolean[] used,int arr[],int n,int position){
        if(position > n){
            count ++;
            return ;
        }
        for(int i=0;i<n;i++){
            if(used[i]){
                continue ;
            }
            if(position % arr[i] == 0 || arr[i] % position == 0){
                used[i] = true;
                back(used,arr,n,position+1);
                used[i] = false ;
            }
        }
    }
}
